cmake_minimum_required(VERSION 3.28)
project(config_consumer LANGUAGES CXX)
if(MODE STREQUAL "install")
  find_package(nlohmann_json CONFIG REQUIRED PATHS "${CONFIG_TEST_PREFIX}" NO_DEFAULT_PATH)
  find_package(config 0.2.2 CONFIG REQUIRED PATHS "${CONFIG_TEST_PREFIX}" NO_DEFAULT_PATH)
else()
  include(FetchContent)
  FetchContent_Declare(json
    GIT_REPOSITORY https://github.com/nlohmann/json.git GIT_TAG v3.12.0)
  FetchContent_MakeAvailable(json)
  if(MODE STREQUAL "parent")
    set(BUILD_TESTING OFF CACHE BOOL "" FORCE)
    set(BUILD_CONFIG_EXAMPLES OFF CACHE BOOL "" FORCE)
    set(BUILD_CONFIG_BENCHMARK OFF CACHE BOOL "" FORCE)
    set(CONFIG_FETCH_JSON OFF CACHE BOOL "" FORCE)
    set(CMAKE_DISABLE_FIND_PACKAGE_nlohmann_json TRUE)
    add_subdirectory("${CONFIG_SOURCE_DIR}" config-library)
  endif()
endif()
add_executable(consumer main.cpp peer.cpp)
if(MODE STREQUAL "manual")
  # 只使用文档要求的头文件、JSON、线程和系统库。
  target_include_directories(consumer PRIVATE "${CONFIG_SOURCE_DIR}/include")
  target_compile_features(consumer PRIVATE cxx_std_20)
  find_package(Threads REQUIRED)
  target_link_libraries(consumer PRIVATE nlohmann_json::nlohmann_json Threads::Threads)
  if(WIN32)
    target_link_libraries(consumer PRIVATE Shell32)
  endif()
else()
  target_link_libraries(consumer PRIVATE config::config)
endif()
if(CMAKE_CXX_COMPILER_ID STREQUAL "MSVC")
  # 显式覆盖中文代码页，避免测试的 UTF-8 选项掩盖公共头的编码问题。
  target_compile_options(consumer PRIVATE /source-charset:.936)
endif()
enable_testing()
add_test(NAME round_trip COMMAND consumer)
