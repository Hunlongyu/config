foreach(example IN ITEMS basic sources observe serialization)
  add_executable(example_${example} ${example}.cpp)
  target_link_libraries(example_${example} PRIVATE config::config)
  config_development_target(example_${example})
  if(BUILD_TESTING)
    file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/run-${example}")
    add_test(NAME example_${example} COMMAND example_${example})
    set_tests_properties(example_${example} PROPERTIES
      WORKING_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/run-${example}" TIMEOUT 15)
  endif()
endforeach()
